const handleBackNavigation = async () => {

  // STEP 1: Get IDs
  const counselorId = resolveCounselorId();
  const chatId = getChatIdForAPI();

  // STEP 2: Check if already rated THIS COUNSELOR
  const alreadyRated = await ratingService.isAlreadyRated(counselorId);
  if (alreadyRated) {
    return false; ✅ Allow navigation, no popup
  }

  // STEP 3: Get all pending ratings from AsyncStorage
  const allPending = await ratingService.getAllPendingRatings();
  // Returns: [{chatId: "abc123", counselorId: "A", ...}, ...]

  // STEP 4: Check if THIS CHAT has pending rating
  const hasPending = allPending.some(r => r.chatId === chatId);

  if (hasPending && !ratingPromptedRef.current) {
    // STEP 5: Show the popup
    setRatingTarget(pendingData);
    setShowRatingModal(true);
    ratingPromptedRef.current = true;
    return true; ❌ Block navigation, show popup
  }

  // STEP 6: No pending rating
  return false; ✅ Allow navigation, no popup
};

useEffect(() => {

  const unsubscribe = navigation.addListener('beforeRemove', async (e) => {
    //        👆 Fired when screen about to be removed
    //           • Hardware back button pressed
    //           • Swipe back gesture
    //           • navigation.goBack() called
    //           • Any navigation away from this screen

    // GUARD 1: Skip if modal already showing
    if (showRatingModal || ratingPromptedRef.current) {
      return; // Don't interfere with existing popup
    }

    // GUARD 2: Call the rating check function
    const shouldBlock = await handleBackNavigation();
    //                  👆 Does all the logic above

    // GUARD 3: If rating pending, prevent navigation
    if (shouldBlock) {
      e.preventDefault(); // ❌ Stop the back action
      // Rating modal is shown by handleBackNavigation()
    }
    // If shouldBlock is false, listener does nothing
    // Navigation continues normally ✅
  });

  // Cleanup: Unsubscribe when component unmounts
  return unsubscribe;

}, [navigation, showRatingModal]);
